# Slice 5: renaming a column across code (refactor_rename_col) on WWI
# a lone column keeps the output name it gave
Sales.Customers.CustomerName	N2	select CustomerName from Sales.Customers	select N2 AS CustomerName from Sales.Customers
Sales.Customers.CustomerName	N2	select c.CustomerName, CustomerName x from Sales.Customers c where CustomerName > '' order by c.CustomerName	select c.N2 AS CustomerName, N2 x from Sales.Customers c where N2 > '' order by c.N2
# a derived table's and a CTE's columns keep their names, so a * over them stays
Sales.Customers.CustomerName	N2	select * from (select CustomerName from Sales.Customers) d	select * from (select N2 AS CustomerName from Sales.Customers) d
Sales.Customers.CustomerName	N2	with c as (select CustomerName from Sales.Customers) select * from c	with c as (select N2 AS CustomerName from Sales.Customers) select * from c
# a subquery's one value and a UNION's later branch have no name to keep
Sales.Customers.CustomerName	N2	select 1 where 'x' in (select CustomerName from Sales.Customers) union select CustomerName from Sales.Customers	select 1 where 'x' in (select N2 from Sales.Customers) union select N2 from Sales.Customers
# a * over the table itself would change its output: refused; under EXISTS it means nothing
Sales.Customers.CustomerName	N2	select * from Sales.Customers	REFUSED: a * includes the column
Sales.Customers.CustomerName	N2	select 1 where exists (select * from Sales.Customers)	select 1 where exists (select * from Sales.Customers)
# written columns, brackets where the new name needs them
Sales.Customers.CustomerName	new name]	update Sales.Customers set CustomerName = CustomerName + 'x' where 1 = 0	update Sales.Customers set [new name]]] = [new name]]] + 'x' where 1 = 0
Sales.Customers.CustomerName	select	insert Sales.Customers (CustomerName) select 'x' where 1 = 0	insert Sales.Customers ([select]) select 'x' where 1 = 0
# a name that may be the column but does not resolve: refused
